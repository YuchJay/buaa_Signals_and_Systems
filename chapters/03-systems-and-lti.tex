\chapter{系统性质与 LTI 结构}
\label{chap:systems-and-lti}

\section{系统的数学描述}
\label{sec:system-description}

\keyterm{系统（system）}接收输入，按固定规则产生输出。画成箭头：
\begin{equation}
  x(t)\longrightarrow\boxed{\text{系统}}\longrightarrow y(t),
  \qquad
  x[n]\longrightarrow\boxed{\text{系统}}\longrightarrow y[n].
  \label{eq:system-flow}
\end{equation}
讨论性质时，看的是\red{框中的固定规则}，不是某组特定的输入输出波形。下文说“$x$ 产生 $y$”，均指同一个系统。

\begin{pitfall}[拿一组输入输出当证据]
  某个输入恰好得到成比例的输出，不能说明线性；某个输入平移后输出也平移，也不能说明时不变。定义要求\blue{对任意允许的输入}都成立。
\end{pitfall}

\section{线性系统}
\label{sec:linear-systems}

\begin{definition}[线性系统]
  \label{def:linear-system}
  设
  \[
    x_1\longrightarrow\boxed{\text{系统}}\longrightarrow y_1,
    \qquad
    x_2\longrightarrow\boxed{\text{系统}}\longrightarrow y_2.
  \]
  若对任意输入 $x_1,x_2$ 和任意常数 $a,b$，都有
  \begin{equation}
    a x_1+b x_2
    \longrightarrow\boxed{\text{同一系统}}\longrightarrow
    a y_1+b y_2,
    \label{eq:linearity-test}
  \end{equation}
  则称该系统为\keyterm{线性系统（linear system）}。这一条件也称为叠加原理。
\end{definition}

可拆成两个条件，缺一不可：
\begin{itemize}
  \item $x\longrightarrow y$ 时 $a x\longrightarrow a y$，叫\keyterm{齐次性（homogeneity）}；
  \item $x_1\longrightarrow y_1$、$x_2\longrightarrow y_2$ 时 $x_1+x_2\longrightarrow y_1+y_2$，叫\keyterm{可加性（additivity）}。
\end{itemize}

\noindent\textbf{快速判据}
\begin{enumerate}
  \item \red{每一项都有输入}，没有与 $x$ 无关的常数项；
  \item \red{每个输入都是一次}，没有平方、乘积、绝对值、指数这类运算。
\end{enumerate}
\hl{先代零输入}：线性系统必须给出零输出：
\begin{equation}
  0\longrightarrow\boxed{\text{线性系统}}\longrightarrow0.
  \label{eq:zero-input-zero-output}
\end{equation}
这条\blue{只能用来否定}：零输入产生非零输出，一定非线性；零输入产生零输出，不能说明线性。

\red{证明非线性只需一个反例；证明线性要对一般的 $ax_1+bx_2$ 成立。}

\begin{example}[增益、微分、积分、差分]
  增益 $y(t)=3x(t)$：
  \[
    3(ax)=a(3x),\qquad
    3(x_1+x_2)=3x_1+3x_2.
  \]
  微分 $y(t)=x'(t)$：
  \[
    \frac{\dd(ax_1+bx_2)}{\dd t}
    =a\frac{\dd x_1}{\dd t}+b\frac{\dd x_2}{\dd t}.
  \]
  积分、差分、累加同理。积分和累加的检验\blue{要求相应的积分、求和存在}。
\end{example}

\begin{example}[常数偏置与平方]
  $y(t)=x(t)+1$：零输入得到 $1$，非线性。

  $y(t)=x^2(t)$：零输入得到零，但
  \[
    2x(t)\longrightarrow4x^2(t)\ne2x^2(t),
  \]
  齐次性不成立，仍然非线性。
\end{example}

只验证“输入相加、输出也相加”不够，系统还必须允许任意缩放。遇到平方、绝对值、指数、输入相乘，\hl{先试齐次性}，通常能迅速发现矛盾。

\section{时不变系统}
\label{sec:time-invariant-systems}

\begin{definition}[时不变系统]
  \label{def:time-invariant-system}
  若
  \[
    x(t)\longrightarrow\boxed{\text{系统}}\longrightarrow y(t),
  \]
  且对任意实数 $t_0$ 都有
  \begin{equation}
    x(t-t_0)\longrightarrow\boxed{\text{同一系统}}\longrightarrow y(t-t_0),
    \label{eq:ct-time-invariance}
  \end{equation}
  则称该系统为\keyterm{时不变系统（time-invariant system）}。离散情形对任意整数 $n_0$：
  \begin{equation}
    x[n-n_0]\longrightarrow\boxed{\text{同一系统}}\longrightarrow y[n-n_0].
    \label{eq:dt-time-invariance}
  \end{equation}
\end{definition}

\noindent\textbf{快速判据}
\begin{enumerate}
  \item \red{$t$ 只在输入括号里}，公式中没有裸露的 $t$；
  \item \red{括号里的 $t$ 只能是 $t$}，不能换成 $2t$、$-t$、$t^2$。
\end{enumerate}
含积分、求和的系统，\blue{仍应按定义验证}。

严格判断走两条路径，\hl{先换输入，再换输出里的时间}：
\begin{enumerate}
  \item 把输入换成 $x(t-t_0)$，代入系统，得到 $y_1(t)$；
  \item 先算原输出 $y(t)$，再把其中\red{每个显含的 $t$} 换成 $t-t_0$，得到 $y_2(t)$。
\end{enumerate}
对任意输入、任意 $t_0$ 都有 $y_1=y_2$，才满足时不变性。离散情形完全类似。

\begin{example}[延迟与非线性的组合]
  $y(t)=x(t-1)$：输入 $x(t-t_0)$ 得到 $x(t-t_0-1)$，恰为 $y(t-t_0)$，时不变。

  $y(t)=\e^{x(t+1)}$：虽然非线性，还用了未来输入，但输入平移后得到
  \[
    \e^{x(t-t_0+1)}=y(t-t_0),
  \]
  仍然时不变。\red{时不变性与线性、因果性相互独立}。
\end{example}

\begin{example}[显含时间与时间缩放]
  $y(t)=t x(t)$：设 $f(t)=x(t-t_0)$，系统给出
  \[
    t x(t-t_0),
  \]
  但 $y(t-t_0)=(t-t_0)x(t-t_0)$，一般不相等，所以时变。

  $y(t)=x(2t)$：两条路径分别为
  \[
    x(2t-t_0),
    \qquad
    x(2t-2t_0),
  \]
  一般也不相等，时变。
\end{example}

取具体输入更直观。对 $y(t)=t x(t)$，令 $x(t)=\stepfun(t)$，延迟 $1$ 后：
\[
  y_1(t)=t\stepfun(t-1),\qquad
  y_2(t)=(t-1)\stepfun(t-1).
\]
在 $t=2$ 处分别是 $2$ 和 $1$。

对 $y(t)=x(2t)$，取 $[0,1]$ 上高度为 $1$ 的矩形。原输出非零区间是 $[0,1/2]$。输入延迟 $1$ 后，$x(2t-1)$ 的非零区间是 $[1/2,1]$；而原输出延迟 $1$ 的非零区间是 $[1,3/2]$，两者不一致。

\red{公式中显含 $t$，或者把 $t$ 换成 $2t$、$-t$，通常会破坏时不变性}。

\section{因果系统}
\label{sec:causal-systems}

\begin{definition}[因果系统]
  \label{def:causal-system}
  若对任意时刻 $t_0$，只要两个输入在所有 $t\le t_0$ 上相同，相应输出在 $t_0$ 也必相同，则称该系统为\keyterm{因果系统（causal system）}。$y(t_0)$ 只取当前和过去的输入，$t>t_0$ 的部分不参与：\red{输出不依赖未来输入}。

  离散情形把条件改成两个输入在所有 $n\le n_0$ 上相同，$y[n_0]$ 只依赖 $x[k]$（$k\le n_0$）。
\end{definition}

\noindent\textbf{判据}
固定输出时刻 $t_0$（或 $n_0$），列出计算该输出所需的输入时刻。直接出现的 $x(\varphi(t_0))$ 要满足
\[
  \varphi(t_0)\le t_0,
  \qquad
  \varphi(n_0)\le n_0.
\]
可概括为：\red{输入括号里的时刻不晚于输出时刻}。积分、求和则看\blue{整个读取区间}；只要有一个严格晚于输出时刻的输入确实影响输出，就非因果。

\begin{example}[延迟、超前与时间缩放]
  $y(t)=x(t-1)$ 只使用过去输入，因果；$y(t)=x(t+1)$ 使用未来输入，非因果。

  $y(t)=x(2t)$：取 $t_0=1$，输出使用 $x(2)$，非因果。
  $y(t)=x(t/2)$：取 $t_0=-1$，输出使用 $x(-1/2)$，仍晚于 $-1$，非因果。

  离散 $y[n]=x[3-n]$：取 $n_0=1$，$y[1]=x[2]$，非因果。
\end{example}

\begin{example}[积分与累加的上限]
  \[
    y(t)=\int_{-\infty}^{t}x(\tau)\dd\tau,
    \qquad
    y[n]=\sum_{k=-\infty}^{n}x[k]
  \]
  只汇总当前和过去输入，因果。把上限改成 $t+1$ 或 $n+1$，当前输出就要使用未来输入，非因果。\hl{先比较上限和观察时刻}，通常最快。
\end{example}

因果性看的是系统映射对所有输入的性质，不是某个输入恰好在未来为零。检查时关注\red{公式可能读取哪些输入时刻}。

\section{无记忆系统}
\label{sec:memoryless-systems}

\begin{definition}[无记忆系统]
  \label{def:memoryless-system}
  若对任意 $t_0$，$y(t_0)$ 只依赖同一时刻的 $x(t_0)$，则称该系统为\keyterm{无记忆系统（memoryless system）}。离散情形是 $y[n_0]$ 只依赖 $x[n_0]$。
\end{definition}

\noindent\textbf{判据}
$x$ 和 $y$ 括号里的自变量\red{完全一样}。连续时间只能出现 $x(t)$，离散时间只能出现 $x[n]$。只要出现 $x(\varphi(t))$ 且 $\varphi(t)\ne t$，就有记忆。

这里的“记忆”只取决于\red{是否需要读取当前时刻以外的输入}，与系统内部有无存储器无关。$y(t)=t x(t)$ 虽然显含时间，仍然无记忆。

\begin{example}[逐点运算与延迟]
  $y(t)=x^2(t)+\e^{x(t)}$ 只使用当前值，无记忆。
  $y(t)=x(t-1)$ 使用过去值，有记忆。
  $y[n]=x^2[n]$ 无记忆；$y[n]=x[2n]$ 在 $n=1$ 处需要使用 $x[2]$，有记忆。
\end{example}

\red{无记忆系统一定因果}：只依赖当前输入，自然不涉及未来。反过来不成立，$y(t)=x(t-1)$ 因果但有记忆。

只检查原点不够。$y[n]=x[2n]$ 在 $n=0$ 处确实只用 $x[0]$，换一个下标便不再成立。\blue{要对所有时刻成立}。

\section{可逆系统}
\label{sec:invertible-systems}

\begin{definition}[可逆系统]
  \label{def:invertible-system}
  若不同输入一定给出不同输出，也就是能由 $y$ 唯一确定 $x$，则称该系统为\keyterm{可逆系统（invertible system）}。这时可在原系统后接\keyterm{逆系统}：
  \begin{equation}
    x\longrightarrow\boxed{\text{原系统}}\longrightarrow y
    \longrightarrow\boxed{\text{逆系统}}\longrightarrow x.
    \label{eq:inverse-system}
  \end{equation}
\end{definition}

\red{能从一般的 $y$ 唯一解出 $x$，就可逆；两个不同输入产生同一输出，就不可逆}。后者通常是举反例的常用方法。

\begin{example}[延迟、平方、累加]
  $y(t)=x(t-1)$：$x(t)=y(t+1)$，可逆。

  $y(t)=x^2(t)$：$x(t)$ 与 $-x(t)$ 给出同一输出，符号信息丢失，不可逆。若\blue{事先限定输入非负}，又可用 $\sqrt{y}$ 恢复——\red{可逆性依赖允许的信号集合}。

  $y[n]=\sum_{k=-\infty}^{n}x[k]$：相邻输出相减得 $x[n]=y[n]-y[n-1]$，可逆。\hl{把恢复公式代回原关系}，可直接验证。
\end{example}

\begin{pitfall}[积分常数]
  $y(t)=x'(t)$ 直接反解成 $x(t)=\int y(t)\dd t$ 会遗漏常数。$x(t)$ 与 $x(t)+C$ 导数相同，所以在一般可微信号上不可逆；\blue{额外给定一个初值}后才能唯一恢复。
\end{pitfall}

\section{稳定系统}
\label{sec:stable-systems}

\begin{definition}[BIBO 稳定系统]
  \label{def:bibo-stable-system}
  若\red{每一个有界输入}都产生有界输出，则称该系统为\keyterm{有界输入有界输出稳定系统（bounded-input bounded-output stable system）}，简称 BIBO 稳定。连续时间的判据：
  \[
    |x(t)|\le M_x<\infty,\ \forall t
    \quad\Longrightarrow\quad
    |y(t)|\le M_y<\infty,\ \forall t,
  \]
  $M_y$ 可以依赖输入和 $M_x$，但\blue{不能随时间增长}。离散把 $t$ 换成 $n$。
\end{definition}

“有界”指存在 $M$，对所有时刻均有 $|x(t)|<M$。

\red{证明稳定要一般，否定稳定用反例}：\hl{从任意界推统一上界}，或者找一个有界输入使输出无界。

\begin{example}[逐点非线性仍然稳定]
  $y(t)=\e^{x(t)}$：$|x(t)|\le M_x$ 时
  \[
    |y(t)|=\e^{x(t)}\le \e^{|x(t)|}\le \e^{M_x},
  \]
  稳定。

  $y(t)=x^3(t)-2x^2(t)+x(t)+1$：
  \[
    |y(t)|\le |x(t)|^3+2|x(t)|^2+|x(t)|+1
    \le M_x^3+2M_x^2+M_x+1,
  \]
  右端是有限常数，也稳定。
\end{example}

\begin{example}[微分、积分、差分]
  理想微分不稳定：有界阶跃的导数含冲激，输出不再是普通有界函数。

  积分器也不稳定：取 $x(t)=\stepfun(t)$，
  \[
    y(t)=\int_{-\infty}^{t}\stepfun(\tau)\dd\tau=t\stepfun(t),
  \]
  随时间无限增长。

  离散差分
  \[
    |x[n]-x[n-1]|\le |x[n]|+|x[n-1]|\le2M_x
  \]
  稳定。系统名称不能决定稳定性，应看它是否会把有界输入\red{累积或放大成无界输出}。
\end{example}

\begin{example}[同一个系统，多项性质分别判断]
  下面三个系统取实输入，微分器限于可微信号。
  \begin{center}
    \small
    \begin{tabular}{lcccccc}
      \toprule
      系统&线性&时不变&因果&无记忆&可逆&BIBO 稳定\\
      \midrule
      $y(t)=\e^{x(t)}$&否&是&是&是&是&是\\
      $y[n]=x[n]x[n-1]$&否&是&是&否&否&是\\
      $y(t)=x'(t)$&是&是&是&否&否&否\\
      \bottomrule
    \end{tabular}
  \end{center}
  $y=\e^x$：只使用当前输入，所以无记忆、因果；延迟输入得 $\e^{x(t-t_0)}$，时不变；$x=\ln y$ 可恢复；上界 $\e^{M_x}$。

  $y[n]=x[n]x[n-1]$：输入乘 $a$，输出乘 $a^2$，非线性；输出相应延迟，时不变；使用当前和前一样值，有记忆；恒为 $1$ 和恒为 $-1$ 都给出恒为 $1$ 的输出，不可逆；$|y|\le M_x^2$，稳定。

  微分器：线性和时不变由求导规则给出；当前导数可用左侧差商计算，因果但有记忆；常数项丢失，不可逆；取 $x(t)=\sin(t^2)$，输出 $2t\cos(t^2)$ 无界，不稳定。
\end{example}
